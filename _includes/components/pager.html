{%- comment -%} Previous / next links across an app's pages. This replaces the old header menu. An app is a short, ordered sequence (overview, then the guides, then any block docs), so stepping through it beats a list of every page repeated on every screen. Order is: the app's overview, then each second-level page alphabetically, with that page's own children immediately after it -- the same order the docs read in, and the same order the homepage cards use. {%- endcomment -%} {%- assign pg_parts = page.url | remove_first: '/' | split: '/' -%} {%- if pg_parts[0] == 'apps' -%} {%- assign pg_app = pg_parts[1] -%} {%- assign pg_prefix = '/apps/' | append: pg_app | append: '/' -%} {%- assign pg_pages = site.html_pages | where_exp: 'p', 'p.url contains pg_prefix' | where_exp: 'p', 'p.title' -%} {%- assign pg_root = pg_pages | where_exp: 'p', 'p.parent == nil' | first -%} {%- if pg_root -%} {%- comment -%} Flatten into reading order. {%- endcomment -%} {%- assign ordered = '' | split: '' -%} {%- assign ordered = ordered | push: pg_root -%} {%- assign pg_children = pg_pages | where_exp: 'p', 'p.parent == pg_root.title' | where_exp: 'p', 'p.grand_parent == nil' | sort: 'nav_order' -%} {%- for child in pg_children -%} {%- assign ordered = ordered | push: child -%} {%- assign pg_gc = pg_pages | where_exp: 'p', 'p.grand_parent == pg_root.title' | where_exp: 'p', 'p.parent == child.title' | sort: 'nav_order' -%} {%- for gc in pg_gc -%} {%- assign ordered = ordered | push: gc -%} {%- endfor -%} {%- endfor -%} {%- assign idx = -1 -%} {%- for item in ordered -%} {%- if item.url == page.url -%}{%- assign idx = forloop.index0 -%}{%- endif -%} {%- endfor -%} {%- if idx >= 0 and ordered.size > 1 -%} {%- assign prev_i = idx | minus: 1 -%} {%- assign next_i = idx | plus: 1 -%}
